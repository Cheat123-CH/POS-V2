services:
  db-pos:
    image: postgres:15-alpine
    container_name: db-pos
    restart: always
    environment:
      POSTGRES_USER: admin
      POSTGRES_PASSWORD: password123
      POSTGRES_DB: pos_v2
    volumes:
      - ./postgres_data:/var/lib/postgresql/data

  api-v4:
    build: ./api-v4
    container_name: api-v4
    restart: always
    depends_on:
      - db-pos

  web-v4:                          # ✅ Same level as db-pos and api-v4
    build:
      context: ./web-v4
      args:
        - API_BASE_URL=https://pos.metsokhcheat.com/api
        - FILE_BASE_URL=https://pos.metsokhcheat.com/file
    container_name: web-v4
    restart: always

  nginx:
    image: nginx:alpine
    container_name: nginx-gateway
    restart: always
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./certbot/conf:/etc/letsencrypt
      - ./certbot/www:/var/www/certbot
    depends_on:
      - api-v4
      - web-v4

  certbot:
    image: certbot/certbot
    volumes:
      - ./certbot/conf:/etc/letsencrypt
      - ./certbot/www:/var/www/certbot
